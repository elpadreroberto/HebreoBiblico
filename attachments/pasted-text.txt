        function renderNiveles() {
            const cont = document.getElementById("niveles");
            cont.innerHTML = "";
            [1,2,3,4,5].forEach(n => {
                const b = document.createElement("button");
                b.className = "px-4 py-2 rounded-lg font-bold transition-all text-sm sm:text-base border border-[#2a3a5a] bg-[#16213e] text-white hover:bg-[#e94560]";

                if (n === nivelActual) b.classList.add("bg-[#e94560]", "shadow-lg");

        function obtenerPartesDelNivel() {
            const dataNivel = nivelesDatos[nivelActual];
            const partes = [];
            let totalPartes = Math.ceil(dataNivel.length / 5);
            
            for(let p = 1; p <= totalPartes; p++) {
                const start = (p - 1) * 5;
                const chunk = dataNivel.slice(start, start + 5);
                partes.push({id: p, label: `${p}`, items: chunk});
            }
            // Examen global al final de cada nivel
            partes.push({id: 'global', label: 'Global', items: [...dataNivel]});
            return partes;
        }

        function renderPartes() {
            const cont = document.getElementById("partes");
            cont.innerHTML = "";
            const partes = obtenerPartesDelNivel();
            
            partes.forEach(pt => {
                const b = document.createElement("button");
                b.className = "px-3 py-1 rounded-md font-bold transition-all text-xs sm:text-sm border border-[#2a3a5a] bg-[#16213e] text-gray-300 hover:text-white";
                b.textContent = pt.label;
                if (pt.id === parteActual) {
                    b.classList.add("bg-[#3be8b0]", "text-[#1a1a2e]", "border-[#3be8b0]", "shadow");
                }
                if (pt.id === 'global') {
                    b.classList.remove('text-gray-300');
                    b.classList.add('text-[#f39c12]');
                    if (pt.id === parteActual) {
                        b.classList.remove("bg-[#3be8b0]");
                        b.classList.add("bg-[#f39c12]");
                    }
                }
                b.onclick = () => setParte(pt.id);
                cont.appendChild(b);
            });
        }

        function setParte(p) {
            parteActual = p;
            renderNiveles();
            renderPartes();
            
            const partes = obtenerPartesDelNivel();
            const encontrada = partes.find(x => x.id === p) || partes[0];
            itemsMuestra = encontrada.items;
            
            items = modo === 'enseñar' ? itemsMuestra : shuffle([...itemsMuestra]);
            indice = 0;
            aciertos = 0;
            mostrar();
        }

        function navegarEstudio(dir) {
            if (modo !== 'enseñar') return;
            indice += dir;
            if (indice < 0) {
                const partes = obtenerPartesDelNivel();
                let idxActual = partes.findIndex(x => x.id === parteActual);
                if (idxActual > 0) {
                    setParte(partes[idxActual - 1].id);
                    indice = items.length - 1;
                } else {
                    indice = 0;
                }
            } else if (indice >= items.length) {
                const partes = obtenerPartesDelNivel();
                let idxActual = partes.findIndex(x => x.id === parteActual);
                if (idxActual < partes.length - 1) {
                    setParte(partes[idxActual + 1].id);
                } else {
                    indice = items.length - 1;
                }
            } else {
                mostrar();
            }
        }

        function mostrar() {

            const he = document.getElementById("hebreo");

        function responder(btn, ok, it) {
            document.querySelectorAll("#opciones button").forEach(b => {
                b.disabled = true;
                b.classList.add("opacity-60");
            });
            btn.classList.remove("opacity-60", "bg-[#0f3460]", "hover:bg-[#1a4b8c]");
            
            if (ok) {
                btn.classList.add("bg-[#2ecc71]", "text-[#1a1a2e]");
                aciertos++;
                document.getElementById("feedback").innerHTML = `<span class="text-[#2ecc71]">¡Correcto!</span>`;
            } else {
                btn.classList.add("bg-[#e74c3c]");
                document.getElementById("feedback").innerHTML = `<span class="text-[#e74c3c]">Incorrecto. Era: ${it.es}</span>`;
                document.querySelectorAll("#opciones button").forEach(b => {
                    if(b.textContent.includes(it.es)) {
                        b.classList.remove("opacity-60", "bg-[#0f3460]");
                        b.classList.add("bg-[#2ecc71]", "text-[#1a1a2e]", "ring-2", "ring-white");
                    }
                });
            }
            
            document.getElementById("detalle").style.display = "block";
            
            setTimeout(() => {
                indice++;
                mostrar();
            }, ok ? 400 : 1500);
        }
